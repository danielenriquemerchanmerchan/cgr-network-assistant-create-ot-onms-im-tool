-- =====================================================
-- TABLA 1: SMARTSOC_SOLICITUDES_USUARIO
-- =====================================================

CREATE TABLE SMARTSOC_SOLICITUDES_USUARIO (
    ID_SOLICITUD NUMBER PRIMARY KEY,
    FECHA_SOLICITUD DATE DEFAULT SYSDATE NOT NULL,
    NOMBRE VARCHAR2(100) NOT NULL,
    APELLIDO VARCHAR2(100) NOT NULL,
    CORREO VARCHAR2(150) NOT NULL,
    AREA VARCHAR2(100) NOT NULL,
    USUARIO_RED VARCHAR2(50) NOT NULL,
    TELEFONO VARCHAR2(20) NOT NULL,
    PASSWORD_SOLICITADO VARCHAR2(100) NOT NULL,
    ESTADO VARCHAR2(20) DEFAULT 'PENDIENTE' CHECK (ESTADO IN ('PENDIENTE', 'APROBADO', 'RECHAZADO')),
    PERFIL_ASIGNADO VARCHAR2(50),
    FECHA_PROCESADO DATE,
    PROCESADO_POR VARCHAR2(50),
    MOTIVO_RECHAZO VARCHAR2(500),
    OBSERVACIONES VARCHAR2(500)
);

CREATE SEQUENCE SMARTSOC_SEQ_SOLICITUDES START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE INDEX SMARTSOC_IDX_SOL_ESTADO ON SMARTSOC_SOLICITUDES_USUARIO(ESTADO);
CREATE INDEX SMARTSOC_IDX_SOL_CORREO ON SMARTSOC_SOLICITUDES_USUARIO(CORREO);
CREATE INDEX SMARTSOC_IDX_SOL_USUARIO ON SMARTSOC_SOLICITUDES_USUARIO(USUARIO_RED);

COMMENT ON TABLE SMARTSOC_SOLICITUDES_USUARIO IS 'Solicitudes de acceso al sistema SmartSOC';


-- =====================================================
-- TABLA 2: SMARTSOC_USUARIOS
-- =====================================================

CREATE TABLE SMARTSOC_USUARIOS (
    ID_USUARIO NUMBER PRIMARY KEY,
    USUARIO VARCHAR2(50) UNIQUE NOT NULL,
    PASSWORD VARCHAR2(200) NOT NULL,
    NOMBRE VARCHAR2(100) NOT NULL,
    APELLIDO VARCHAR2(100) NOT NULL,
    CORREO VARCHAR2(150) UNIQUE NOT NULL,
    AREA VARCHAR2(100),
    TELEFONO VARCHAR2(20),
    PERFIL VARCHAR2(50) NOT NULL,
    FECHA_CREACION DATE DEFAULT SYSDATE NOT NULL,
    FECHA_ULTIMA_MODIFICACION DATE,
    ULTIMO_ACCESO DATE,
    ACTIVO NUMBER(1) DEFAULT 1 CHECK (ACTIVO IN (0, 1)),
    CREADO_POR VARCHAR2(50),
    ID_SOLICITUD_ORIGEN NUMBER,
    CONSTRAINT SMARTSOC_FK_USUARIO_SOLIC FOREIGN KEY (ID_SOLICITUD_ORIGEN) 
        REFERENCES SMARTSOC_SOLICITUDES_USUARIO(ID_SOLICITUD)
);

CREATE SEQUENCE SMARTSOC_SEQ_USUARIOS START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE INDEX SMARTSOC_IDX_USU_USUARIO ON SMARTSOC_USUARIOS(USUARIO);
CREATE INDEX SMARTSOC_IDX_USU_CORREO ON SMARTSOC_USUARIOS(CORREO);
CREATE INDEX SMARTSOC_IDX_USU_PERFIL ON SMARTSOC_USUARIOS(PERFIL);
CREATE INDEX SMARTSOC_IDX_USU_ACTIVO ON SMARTSOC_USUARIOS(ACTIVO);

COMMENT ON TABLE SMARTSOC_USUARIOS IS 'Usuarios aprobados y activos del sistema SmartSOC';


-- =====================================================
-- TABLA 3: SMARTSOC_AUDITORIA_ACCESOS
-- =====================================================

CREATE TABLE SMARTSOC_AUDITORIA_ACCESOS (
    ID_ACCESO NUMBER PRIMARY KEY,
    USUARIO VARCHAR2(50) NOT NULL,
    PAGINA VARCHAR2(100),
    ACCION VARCHAR2(100),
    FECHA_ACCESO TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    IP_ADDRESS VARCHAR2(50),
    DETALLES VARCHAR2(500)
);

CREATE SEQUENCE SMARTSOC_SEQ_AUDITORIA START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE INDEX SMARTSOC_IDX_AUD_USUARIO ON SMARTSOC_AUDITORIA_ACCESOS(USUARIO);
CREATE INDEX SMARTSOC_IDX_AUD_FECHA ON SMARTSOC_AUDITORIA_ACCESOS(FECHA_ACCESO);

COMMENT ON TABLE SMARTSOC_AUDITORIA_ACCESOS IS 'Registro de auditoría del sistema SmartSOC';
